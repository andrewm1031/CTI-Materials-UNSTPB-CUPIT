PImage img;
void setup(){
   img=loadImage("logan.jpg");
   size(1200,700);
   img.resize(img.width/2,img.height/2);
   for(int i=0;i<img.width;i++)
     for(int j=0;j<img.height;j++)
       {
         if(brightness(img.get(i,j))<120) img.set(i, j, color(0)); else img.set(i, j, color(255));         
       }
}

void draw(){
  image(img,0,0);
}
