int x1, y1, x2, y2, x3, y3, x4, y4;
int xc, yc;
int i;

void setup(){
  size(1000,800);
  i=0;
  frameRate(100);
}

void draw(){
  if(i>=3) {
    int n=1+floor(random(3)); // n poate sa fie 1, 2 sau 3
    if(n==1) {
       xc=(xc+x1)/2;
       yc=(yc+y1)/2;
    }
    if(n==2) {
       xc=(xc+x2)/2;
       yc=(yc+y2)/2;
    }
    if(n==3) {
       xc=(xc+x3)/2;
       yc=(yc+y3)/2;
    }

    circle(xc,yc,5);
  }
}

void mousePressed(){
  i++;
  if(i==1) {
    x1=mouseX;
    y1=mouseY;
    circle(x1,y1,5);
  }
  if(i==2){
    x2=mouseX;
    y2=mouseY;
    circle(x2,y2,5);
  }
  if(i==3){
    x3=mouseX;
    y3=mouseY;
    circle(x3,y3,5);
  }
}

void keyPressed(){
   if(key=='r') fill(255,0,0);
   if(key=='b') fill(0,0,255);
   if(key=='g') fill(0,255,0);
   if(key==' ') {
      background(200);
      i=0;
   }
}
